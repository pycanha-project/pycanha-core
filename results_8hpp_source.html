13// Every sparse radiative result is a row-major Eigen sparse matrix: the same
14// type the TMM coupling matrices use, and the one nanobind converts straight
15// to scipy.sparse.csr_matrix without a hand-written binding.
16using SparseMatrix = Eigen::SparseMatrix<double, Eigen::RowMajor>;
17// Column/row-pointer type of the CSR arrays above (Eigen's default `int`).
18using SparseIndex = SparseMatrix::StorageIndex;

20// Matrix results carry three VIRTUAL bucket columns appended after the
21// num_face_slots real columns, so every parcel of emitted energy has an
22// explicit, node-mappable destination:
23// - space: rays/energy that escaped the scene,
24// - inactive: energy absorbed at inactive (non-radiative) faces
25// (exchange only — the geometric vf kernel scores inactive
26// faces at their own column),
27// - lost: the purely mathematical residue (Russian-roulette balance,
28// max_bounces cutoff). Zero-mean adjustments make this entry
29// SIGNED and it may come out slightly negative.
30// Full rows therefore account for everything: a vf row sums to exactly 1
31// and an exchange row conserves energy exactly.
32inline constexpr std::int64_t num_virtual_columns = 3;
33inline constexpr std::int64_t space_column_offset = 0;
34inline constexpr std::int64_t inactive_column_offset = 1;
35inline constexpr std::int64_t lost_column_offset = 2;

37// Every result carries its own statistics so callers can judge Monte-Carlo
38// convergence without re-tracing.
39struct TraceStats {
40 std::uint64_t total_rays = 0;
41 // Cumulative across accumulate() calls.
42 std::uint64_t rays_per_face = 0;
43 // Per-entry standard-error estimates.
44 double mean_stderr = 0.0;
45 double max_stderr = 0.0;
46 // Matrix results: the largest normalized disagreement of the two
47 // directions of a face pair, measured on the RAW estimates before they
48 // are combined. Measured after combining it would be identically zero
49 // by construction and would stop being the winding/parity check it
50 // exists to be.
51 double reciprocity_residual = 0.0;
52 // Exchange only: energy killed below the threshold.
53 double lost_energy_fraction = 0.0;
54 std::chrono::nanoseconds gpu_time{0};
55};

57// Matrix rows are face slots (global, both sides); columns are the same
58// slots plus the virtual bucket columns above (cols == rows +
59// num_virtual_columns). The consumer maps the buckets to real nodes (e.g.
60// the space node) or drops them.
61//
62// The stored value is the SYMMETRIC, EXTENSIVE quantity
63//
64// G_ij = A_i F_ij = A_j F_ji (units m^2)
65//
66// and only the UPPER TRIANGLE (j >= i) of the real face columns is kept,
67// matching the TMM coupling convention. Storing G rather than F is what
68// makes that lossless: both view factors come back as F_ij = G_ij/A_i and
69// F_ji = G_ij/A_j from the face areas the scene already owns, whereas an
70// upper-triangular F would silently discard one direction. The diagonal is
71// retained even though a planar face slot cannot see itself, so nothing is
72// dropped without saying so. The three bucket columns are always > i, have
73// no transpose partner, and pass through untriangulated.
74struct VfResult {
75 SparseMatrix vf;
76 // Per-row sum over ALL columns of the RAW estimate, before combining
77 // and before thresholding (== 1 exactly for rows that emitted). This is
78 // closure, and it is deliberately not the row sum of `vf`: after
79 // triangulation the stored triangle's row sums mean nothing of the
80 // kind.
81 Eigen::VectorXd row_sums;
82 // The raw, untriangulated G with BOTH triangles, present only when
83 // TriangulationConfig::keep_full_matrix asked for it. Entry (i, j)
84 // holds A_i F_ij and entry (j, i) holds A_j F_ji, so the two
85 // independent estimates of the same coupling can be read off and
86 // compared directly.
87 std::optional<SparseMatrix> full_vf;
88 TraceStats stats;
89};

91// Same shape and the same convention as VfResult: rows are face slots,
92// columns are the slots plus the virtual bucket columns, and only the UPPER
93// TRIANGLE of the real face columns is kept.
94//
95// The stored value is the SYMMETRIC, EXTENSIVE quantity
96//
97// H_ij = A_i eps_i B_ij = A_j eps_j B_ji (units m^2)
98//
99// where B is the Gebhart absorption factor the kernel traces (deposited
100// energy per unit emitted) and eps is the traced band's absorptivity. H is
101// what actually transports heat and what the node-level GR is built from, so
102// storing it is what makes the upper triangle lossless: both factors come
103// back as B_ij = H_ij/(A_i eps_i) and B_ji = H_ij/(A_j eps_j).
104//
105// A slot with eps = 0 in the traced band is the one case where that does not
106// invert. It needs no special handling because there is nothing to recover:
107// such a face absorbs nothing, so B_ji = 0 for every j, and it emits nothing,
108// so its row transports no heat whatever the kernel's unit-energy rays did.
109// Both directions vanish consistently and H stays reciprocal. `full_factors`
110// keeps the raw B for anyone who wants the geometry behind that zero.
111struct ExchangeResult {
112 Band band = Band::IR;
113 SparseMatrix factors;
114 // The raw, untriangulated INTENSIVE B with BOTH triangles, present only
115 // when TriangulationConfig::keep_full_matrix asked for it. Deliberately
116 // not the extensive H that `full_vf` would suggest: for view factors the
117 // intensive form is always recoverable because areas are positive, for
118 // exchange it is not, so the debugging matrix carries the form that
119 // cannot be reconstructed.
120 std::optional<SparseMatrix> full_factors;
121 TraceStats stats;
122};
