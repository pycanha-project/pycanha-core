12// Matrix results carry three VIRTUAL bucket columns appended after the
13// num_face_slots real columns, so every parcel of emitted energy has an
14// explicit, node-mappable destination:
15// - space: rays/energy that escaped the scene,
16// - inactive: energy absorbed at inactive (non-radiative) faces
17// (exchange only — the geometric vf kernel scores inactive
18// faces at their own column),
19// - lost: the purely mathematical residue (Russian-roulette balance,
20// max_bounces cutoff). Zero-mean adjustments make this entry
21// SIGNED and it may come out slightly negative.
22// Full rows therefore account for everything: a vf row sums to exactly 1
23// and an exchange row conserves energy exactly.
24inline constexpr std::int64_t num_virtual_columns = 3;
25inline constexpr std::int64_t space_column_offset = 0;
26inline constexpr std::int64_t inactive_column_offset = 1;
27inline constexpr std::int64_t lost_column_offset = 2;
28
29// Every result carries its own statistics so callers can judge Monte-Carlo
30// convergence without re-tracing.
31struct TraceStats {
32 std::uint64_t total_rays = 0;
33 // Cumulative across accumulate() calls.
34 std::uint64_t rays_per_face = 0;
35 // Per-entry standard-error estimates.
36 double mean_stderr = 0.0;
37 double max_stderr = 0.0;
38 // VF only: max |Ai*Fij - Aj*Fji| (normalized).
39 double reciprocity_residual = 0.0;
40 // Exchange only: energy killed below the threshold.
41 double lost_energy_fraction = 0.0;
42 std::chrono::nanoseconds gpu_time{0};
43};

45// Matrix rows are face slots (global, both sides); columns are the same
46// slots plus the virtual bucket columns above (cols == rows +
47// num_virtual_columns). A vf row including its space column sums to
48// exactly 1; the consumer maps the buckets to real nodes (e.g. the space
49// node) or drops them.
50struct VfResult {
51 SparseF64 vf;
52 // Per-row sum over ALL columns (== 1 exactly for rows that emitted).
53 Eigen::VectorXd row_sums;
54 TraceStats stats;
55};
