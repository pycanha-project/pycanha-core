122 // A coupling position in one block, compared in storage order.
123 struct BlockPosition {
124 StorageIndex row = -1;
125 StorageIndex col = -1;
126
127 [[nodiscard]] bool operator<(
128 const BlockPosition& other) const noexcept {
129 return row < other.row || (row == other.row && col < other.col);
130 }
131 };

183 // The general bulk path of Couplings::add_couplings, block by block:
184 // every valid entry is first planned (to learn whether a block's entries
185 // come in storage order after what it holds), then put: appended straight
186 // into the block when they do, collected and merged in place otherwise.
187 class BulkWriter {

189 explicit BulkWriter(CouplingMatrices& matrices) noexcept

194 // Closes the appends and runs the merges; returns the entries taken.
195 std::size_t finish(std::span<const double> values, CouplingMerge merge,
196 BulkReport& report);
