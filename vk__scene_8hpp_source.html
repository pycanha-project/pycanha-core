28// Minimal RAII-by-owner buffer: created/destroyed by SceneImpl helpers.
29struct GpuBuffer {
30 VkBuffer buffer = VK_NULL_HANDLE;
31 VmaAllocation allocation = VK_NULL_HANDLE;
32 void* mapped = nullptr; // non-null only for host-visible buffers
33 VkDeviceSize size = 0;
34};

36// Host-visible buffers are created with the mapped flag; a null mapping
37// would be a logic error, and checking makes that explicit.
38[[nodiscard]] inline void* checked_mapped(const GpuBuffer& buffer) {
39 if (buffer.mapped == nullptr) {
40 throw std::logic_error("pycanha::radiative: buffer is not host-mapped");
41 }
42 return buffer.mapped;
43}

45// C++ mirror of the InstanceData struct in kernels/common.slang — must
46// match byte-for-byte (scalar layout, 80 bytes). tf_rows are rotation rows
47// with the translation in .w (world = R * p + t).
