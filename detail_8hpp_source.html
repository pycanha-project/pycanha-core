47// Inverts the planar bilinear patch
48// P(u,v) = p1 + u*e + v*f + u*v*g, e = p2-p1, f = p4-p1, g = p1-p2+p3-p4
49// for a point assumed to lie in the patch's plane. Substituting and crossing
50// with (f + u*g) removes v and leaves a quadratic in u:
51// (e x g) u^2 + [(e x f) - (q x g)] u - (q x f) = 0, q = P - p1
52// where x is the 2D cross product in the patch plane. g vanishes for a
53// parallelogram, which degrades the quadratic to the linear solution.
54// Once u is known, v follows from a single projection.
55[[nodiscard]] inline Point2D invert_bilinear(const Vector3D& q,
56 const Vector3D& e,
57 const Vector3D& f,
58 const Vector3D& g,
59 const Vector3D& normal) {
60 const auto cross2d = [&normal](const Vector3D& lhs, const Vector3D& rhs) {
61 return lhs.cross(rhs).dot(normal);
62 };
63
64 const double quad_a = cross2d(e, g);
65 const double quad_b = cross2d(e, f) - cross2d(q, g);
66 const double quad_c = -cross2d(q, f);
67
68 double u = 0.0;
69 if (std::abs(quad_a) <= LENGTH_TOL * LENGTH_TOL) {
70 u = std::abs(quad_b) > LENGTH_TOL * LENGTH_TOL ? -quad_c / quad_b : 0.0;
71 } else {
72 const double discriminant =
73 std::max((quad_b * quad_b) - (4.0 * quad_a * quad_c), 0.0);
74 const double root = std::sqrt(discriminant);
75 // Both roots are real for a convex patch; the one inside [0,1] is the
76 // point's own parameter, the other lies on the patch's other branch.
77 const double u_minus = (-quad_b - root) / (2.0 * quad_a);
78 const double u_plus = (-quad_b + root) / (2.0 * quad_a);
79 const auto outside = [](double value) {
80 return std::max({0.0 - value, value - 1.0, 0.0});
81 };
82 u = outside(u_minus) <= outside(u_plus) ? u_minus : u_plus;
83 }
84
85 const Vector3D v_direction = f + (u * g);
86 const double v_scale = v_direction.squaredNorm();
87 const double v = v_scale > LENGTH_TOL * LENGTH_TOL
88 ? (q - (u * e)).dot(v_direction) / v_scale
89 : 0.0;
90 return {u, v};
91}

102[[nodiscard]] inline double wrap_angle_positive(double angle) noexcept {
103 constexpr double full_turn = std::numbers::pi * 2.0;
104 while (angle < 0.0) {
105 angle += full_turn;

107 while (angle >= full_turn) {
108 angle -= full_turn;
