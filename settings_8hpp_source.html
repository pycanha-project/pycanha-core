7// Monte-Carlo trace control: a fixed ray count plus a seed make every run
8// deterministic and reproducible. Accuracy-targeted convergence is
9// intentionally NOT a C++ concept — the accumulate_* calls are
10// batch-additive, so callers re-invoke until the reported statistics
11// satisfy them.
12struct TraceSettings {
13 // Rays per emitting face, per accumulate() call.
14 std::uint64_t rays_per_face = 10'000;
15 // Deterministic seed; batches use seed + batch_index.
16 std::uint32_t seed = 0;
17 // MCRT ray-kill energy cutoff (exchange kernels).
18 float energy_threshold = 1e-4F;
19 // Hard safety bound on the bounce loop.
20 std::uint32_t max_bounces = 64;
21 // Emit along the face normal instead of the cosine-weighted hemisphere
22 // (the legacy `_Nodes` debugging mode; vf/exchange kernels only).
23 bool normal_emission = false;
24};

26// Accumulator buffer layout. Kernels are identical in both layouts and cells
27// accumulate as integers (integer adds are associative, unlike float adds),
28// so Dense and Tiled results are bit-identical for the same seed.
29enum class AccumLayout : std::uint8_t {
30 Dense, // one Nf x Nf buffer — the small-model fast path
31 Tiled, // tile_rows x Nf blocks streamed + CPU-sparsified — the big path
32};

34// How the two independent Monte-Carlo estimates of a face pair are combined
35// into the single reciprocity-consistent value that gets stored.
36enum class TriangulationMode : std::uint8_t {
37 // Keep the forward estimate as traced. Nothing is combined, so the
38 // stored triangle is still two independent noisy numbers per pair — the
39 // mode to reach for when bisecting a suspected assembly bug.
40 None,
41 // Weight each direction by how densely it was sampled:
42 // X_i = 1/2 (1 + sign(Y) |Y|^n), Y = (v - u) / (v + u)
43 // with u = A_i/N_i and v = A_j/N_j, the two estimator variances up to a
44 // factor that cancels. n = 1 makes this exactly inverse-variance
45 // (minimum-variance) weighting; the default n = 0.4 is the more
46 // aggressive, empirically tuned rule the reference implementation uses.
47 //
48 // The SAME proxy applies to the exchange kernel even though that kernel
49 // deposits energy rather than unit hits: its per-cell variance carries a
50 // further factor eps_i eps_j, which appears in both directions and
51 // therefore cancels out of the weight along with everything else.
52 RayDensity,
53 // Impose reciprocity AND row closure together instead of one after the
54 // other. RayDensity blends each pair independently, which leaves rows no
55 // longer summing to the energy they emitted; renormalising afterwards
56 // then partly undoes the reciprocity. This mode solves for the matrix
57 // closest to the raw estimate in the inverse-variance metric among those
58 // that close every row exactly, which is a sparse symmetric system of
59 // one equation per face slot.
60 //
61 // It ignores `exponent`: the combination it starts from has to be the
62 // unconstrained minimum-variance one (n = 1) or the result is not the
63 // least-squares solution of anything.
64 ConstrainedLeastSquares,
65};

67struct TriangulationConfig {
68 TriangulationMode mode = TriangulationMode::RayDensity;
69 // Must be > 0. 1.0 reduces the weight to inverse-variance weighting,
70 // which measured marginally better than 0.4 on the exchange path (where
71 // the ray-density proxy is exact rather than empirical) — 0.4 is kept as
72 // the single default for both paths so that a default-constructed config
73 // never means two different things.
74 double exponent = 0.4;
75 // Also keep the raw, untriangulated matrix (both triangles) alongside
76 // the combined upper triangle. Off at every model size: it doubles the
77 // result memory and exists only for someone debugging a model.
78 bool keep_full_matrix = false;
79};

81struct AccumConfig {
82 AccumLayout layout = AccumLayout::Dense;
83 // Tiled only; 0 = invalid, must be set (the Python policy derives it).
84 std::uint32_t tile_rows = 0;
85 // Entries with |value| <= threshold are dropped from the result CSR;
86 // row sums and every other statistic are computed BEFORE thresholding,
87 // so closure/conservation accounting stays exact. 0 keeps any nonzero.
88 //
89 // Both matrices compare on the INTENSIVE value: max(F_ij, F_ji) is the
90 // stored G_ij over the smaller face area, max(B_ij, B_ji) the stored
91 // H_ij over the smaller A*eps. A pair is therefore dropped only when
92 // both directions are negligible. That comparison happens AFTER the two
93 // directions are combined: dropping them independently first would leave
94 // a surviving entry combined against a zero, which re-breaks the
95 // reciprocity just imposed and biases the survivor low.
96 double sparse_threshold = 0.0;
97 // Applies to the VF and the exchange accumulator alike.
98 TriangulationConfig triangulation;
99};
