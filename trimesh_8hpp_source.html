13// Triangular surface mesh templated on the vertex scalar type.
14// - TriMeshD (double) is the precise pipeline type used everywhere
15// internally (per-item caches, per-cut-group caches, the model root mesh
16// before conversion).
17// - TriMeshF (float) is what the model root exposes via
18// GeometryModel::mesh().
19//
20// face_ids index the per-side face identity: even = side 1, odd = side 2.
21// Only even ids are ever stored: a triangle always defines BOTH faces of its
22// pair (the odd face is the same triangle with the normal reversed), so the
23// triangle carries the even id and its winding normal is side 1.
24// node_numbers is DENSE, indexed directly by face_id (gaps allowed after
25// cuts): node_numbers[face_id] is the side-1 node, node_numbers[face_id + 1]
26// the side-2 node. primitives records which contiguous face_id range came from
27// which source GeometryId, sorted by first_face_id.
28template <class Scalar>
29class TriMesh {
30 public:
31 using VertexMatrix = Eigen::Matrix<Scalar, Eigen::Dynamic, 3>;
32 using TriangleMatrix = Eigen::Matrix<pycanha::MeshIndex, Eigen::Dynamic, 3>;
33 using FaceIdVector = Eigen::Matrix<pycanha::MeshIndex, Eigen::Dynamic, 1>;
34 using NodeNumberVector = Eigen::Matrix<pycanha::NodeNum, Eigen::Dynamic, 1>;
35
36 struct PrimitiveRange {
37 GeometryId geometry_id{};
38 pycanha::MeshIndex first_face_id = 0; // inclusive
39 pycanha::MeshIndex last_face_id = 0; // inclusive
40 };
41
42 VertexMatrix vertices; // Np x 3
43 TriangleMatrix triangles; // Nt x 3
44 FaceIdVector face_ids; // Nt
45 NodeNumberVector node_numbers; // Nf (dense, indexed by face_id)
46 std::vector<PrimitiveRange> primitives; // sorted by first_face_id
47
48 // How many faces this mesh OWNS, which is not the same as how many still
49 // have triangles: a cut removes triangles but never removes the faces they
50 // belonged to. Set by whoever builds the mesh (the mesher from the
51 // ThermalMesh, the cut backend from its target, concatenation from the sum
52 // of its pieces) and left alone by anything that only drops triangles.
53 // Deriving it from max(face_ids) instead would shrink an item whose LAST
54 // pair was cut away, renumbering every later item in the model.
55 pycanha::MeshIndex num_faces = 0;
56
57 [[nodiscard]] pycanha::MeshIndex np() const noexcept {
58 return static_cast<pycanha::MeshIndex>(vertices.rows());
59 }
60 [[nodiscard]] pycanha::MeshIndex nt() const noexcept {
61 return static_cast<pycanha::MeshIndex>(triangles.rows());
62 }
63 // Number of faces this mesh owns (both sides counted), including faces a
64 // cut left with no triangles of their own.
65 [[nodiscard]] pycanha::MeshIndex nf() const noexcept { return num_faces; }
66
67 // Converts this mesh to a different vertex scalar type. Integer data and
68 // primitive provenance are copied verbatim; only the vertices are cast.
69 template <class Other>
70 [[nodiscard]] TriMesh<Other> cast() const {
71 TriMesh<Other> out;
72 out.vertices = vertices.template cast<Other>();
73 out.triangles = triangles;
74 out.face_ids = face_ids;
75 out.node_numbers = node_numbers;
76 out.num_faces = num_faces;
77 out.primitives.reserve(primitives.size());
78 for (const auto& range : primitives) {
79 out.primitives.push_back(typename TriMesh<Other>::PrimitiveRange{
80 range.geometry_id, range.first_face_id, range.last_face_id});
81 }
82 return out;
83 }
84};

87using TriMeshF = TriMesh<float>;
