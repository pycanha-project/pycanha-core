32// Minimal RAII-by-owner buffer: created/destroyed by SceneImpl helpers.
33// Every buffer is MTLStorageModeShared — on the unified memory of Apple
34// Silicon that one storage mode serves both the staging and the device-local
35// role, so there is no separate host-visible flag as on the Vulkan side.
36struct GpuBuffer {
37 id<MTLBuffer> buffer = nil;
38 std::size_t size = 0;
39};

41// Shared-storage buffers always have host-visible contents; a null mapping
42// would be a logic error, and checking makes that explicit. Defined out of
43// line so this header holds no Objective-C message sends: the repository's
44// clang-format and cpplint checks read .hpp files as plain C++, where a
45// message send parses as something else entirely.
46[[nodiscard]] void* checked_mapped(const GpuBuffer& buffer);

48// C++ mirror of the InstanceData struct in kernels/common.slang — must
49// match byte-for-byte (80 bytes). tf_rows are rotation rows with the
50// translation in .w (world = R * p + t).
51struct InstanceDataGpu {
52 std::uint64_t vertices_addr;
53 std::uint64_t indices_addr;
54 std::uint64_t tri_face_addr;
55 std::uint32_t part_id;
56 std::uint32_t flags;
57 std::array<std::array<float, 4>, 3> tf_rows;
58};
59static_assert(sizeof(InstanceDataGpu) == 80);
